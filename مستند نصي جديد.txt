    <script>
        // تخزين البيانات ديناميكياً لتحديث القوائم المنسدلة تلقائياً

        let banksList = ["البنك الأهلي السعودي", "مصرف الراجحي", "خزينة الإدارة (نقداً)"];

        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById('tab-' + tabId).classList.remove('hidden');

            document.querySelectorAll('.menu-item').forEach(el => {
                el.classList.remove('bg-emerald-700', 'text-white', 'font-bold', 'shadow-sm');
                el.classList.add('hover:bg-emerald-900/60', 'text-emerald-100', 'font-medium');
            });
            const activeMenu = document.querySelector(`[data-tab="${tabId}"]`);
            if (activeMenu) {
                activeMenu.classList.remove('hover:bg-emerald-900/60', 'text-emerald-100', 'font-medium');
                activeMenu.classList.add('bg-emerald-700', 'text-white', 'font-bold', 'shadow-sm');
                document.getElementById('current-page-title').innerText = activeMenu.querySelector('span').innerText;
            }
        }

        function toggleSidebar() {
            const sidebar = document.getElementById('sidebar');
            sidebar.classList.toggle('w-72');
            sidebar.classList.toggle('w-0');
            sidebar.classList.toggle('overflow-hidden');
        }

        function openModal(modalId) {
            document.getElementById(modalId).classList.remove('hidden');
        }
        function closeModal(modalId) {
            document.getElementById(modalId).classList.add('hidden');
        }

        function searchTable(inputId, tableBodyId) {
            const input = document.getElementById(inputId).value.toLowerCase();
            const rows = document.querySelectorAll(`#${tableBodyId} tr`);
            rows.forEach(row => {
                const text = row.innerText.toLowerCase();
                row.style.display = text.includes(input) ? '' : 'none';
            });
        }

        function updateDropdowns() {
            // تحديث قائمة الشركات في نموذج الموظف
            const compSelect = document.getElementById('empCompany');
            compSelect.innerHTML = companiesList.map(c => `<option value="${c}">${c}</option>`).join('');

            // تحديث قائمة الأقسام في نموذج الموظف
            const deptSelect = document.getElementById('empDepartment');
            deptSelect.innerHTML = departmentsList.map(d => `<option value="${d}">${d}</option>`).join('');

            // تحديث قائمة البنوك في نموذج الموظف
            const bankSelect = document.getElementById('empTransfer');
            bankSelect.innerHTML = banksList.map(b => `<option value="${b}">${b}</option>`).join('');
        }

        function addEmployee(event) {
            event.preventDefault();
            const id = document.getElementById('empId').value;
            const name = document.getElementById('empName').value;
            const company = document.getElementById('empCompany').value;
            const department = document.getElementById('empDepartment').value;
            const basic = parseFloat(document.getElementById('empBasic').value) || 0;
            const housing = parseFloat(document.getElementById('empHousing').value) || 0;
            const transport = parseFloat(document.getElementById('empTransport').value) || 0;
            const presidency = parseFloat(document.getElementById('empPresidency').value) || 0;
            const management = parseFloat(document.getElementById('empManagement').value) || 0;
            const transfer = document.getElementById('empTransfer').value;

            const totalAllowances = housing + transport + presidency + management;
            const totalSalary = basic + totalAllowances;

            // إضافة الجدول الرئيسي للموظفين
            const tbody = document.getElementById('employeeTableBody');
            const row = document.createElement('tr');
            row.className = 'hover:bg-gray-50/50';
            row.innerHTML = `
                <td class="p-3 font-bold text-emerald-700">${id}</td>
                <td class="p-3 font-bold text-gray-800">${name}</td>
                <td class="p-3">${company}</td>
                <td class="p-3">${department}</td>
                <td class="p-3">${basic.toLocaleString()}</td>
                <td class="p-3">${housing.toLocaleString()}</td>
                <td class="p-3">${transport.toLocaleString()}</td>
                <td class="p-3">${presidency.toLocaleString()}</td>
                <td class="p-3">${management.toLocaleString()}</td>
                <td class="p-3 font-black text-emerald-700">${totalSalary.toLocaleString()} ر.س</td>
                <td class="p-3"><span class="bg-blue-100 text-blue-800 text-xs px-2.5 py-1 rounded-full font-bold">${transfer}</span></td>
            `;
            tbody.prepend(row);

            // إضافة لمسيّرات الرواتب أيضاً
            const payrollTbody = document.getElementById('payrollTableBody');
            const payRow = document.createElement('tr');
            payRow.innerHTML = `
                <td class="p-3 font-bold text-emerald-700">${id}</td>
                <td class="p-3 font-bold">${name}</td>
                <td class="p-3">${company}</td>
                <td class="p-3">${basic.toLocaleString()}</td>
                <td class="p-3">${totalAllowances.toLocaleString()}</td>
                <td class="p-3 font-black text-emerald-700">${totalSalary.toLocaleString()} ر.س</td>
                <td class="p-3">${transfer}</td>
            `;
            payrollTbody.prepend(payRow);

            // تحديث عداد الإحصائيات
            const countEl = document.getElementById('stat-emp-count');
            countEl.innerText = parseInt(countEl.innerText) + 1;

            closeModal('employeeModal');
            document.getElementById('employeeForm').reset();
            alert('تم إضافة الموظف بنجاح وحفظه في النظام');
        }
